{{- /* One interface glyph, inlined from assets/svg/ui/.svg as it is: the stylesheet hangs off the classes and sizes each file carries (feather-*, meta-icon, the 18px RSS one), so a file is stored exactly as it rendered inline. A glyph is named by the theme itself, never by a site, so a missing one is a bug in the theme and stops the build. chomp: the file's trailing newline would otherwise reach the page. */ -}} {{- $name := lower (trim (printf "%v" (.name | default "")) " \t\r\n") -}} {{- if findRE `^[a-z0-9][a-z0-9-]*$` $name -}} {{- with resources.Get (printf "svg/ui/%s.svg" $name) -}} {{- .Content | chomp | safeHTML -}} {{- else -}} {{- errorf "icon.html: assets/svg/ui/%s.svg does not exist" $name -}} {{- end -}} {{- else -}} {{- errorf "icon.html: %q is not an icon name" $name -}} {{- end -}}
