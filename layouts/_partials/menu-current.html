{{- /* aria-current of a main menu entry: "page", "true" when it holds the page, or "". The decoded url is compared too, without a trailing slash or the index.html of uglyURLs: url = "posts/" links no page. */ -}} {{- $trim := `(?:/index\.html|/)$` -}} {{- $site := urls.Parse site.BaseURL -}} {{- $page := .page -}} {{- $entry := .entry -}} {{- $state := "" -}} {{- $here := $page.RelPermalink -}} {{- with (try (urls.Parse $here)).Value }}{{ $here = .Path }}{{ end -}} {{- $here = replaceRE $trim "" $here -}} {{- $url := "" -}} {{- $local := false -}} {{- if $entry.URL -}} {{- with (try (urls.Parse (partial "menu-url.html" $entry))).Value -}} {{- /* Same host: compared by path. */ -}} {{- if and (not (or .RawQuery .Fragment)) (or (not (or .Scheme .Host)) (eq .Host $site.Host)) -}} {{- $url = replaceRE $trim "" .Path -}} {{- $local = true -}} {{- end -}} {{- end -}} {{- end -}} {{- if or ($page.IsMenuCurrent "main" $entry) (and $local (eq $url $here)) -}} {{- $state = "page" -}} {{- else if $page.HasMenuCurrent "main" $entry -}} {{- $state = "true" -}} {{- /* A home is never an ancestor. */ -}} {{- else if and $local (not (in (slice (replaceRE $trim "" (urls.Parse site.Home.RelPermalink).Path) (replaceRE $trim "" $site.Path)) $url)) (hasPrefix $here (printf "%s/" $url)) -}} {{- $state = "true" -}} {{- else -}} {{- range $entry.Children -}} {{- if partial "menu-current.html" (dict "page" $page "entry" .) -}} {{- $state = "true" -}} {{- end -}} {{- end -}} {{- end -}} {{- return $state -}}
