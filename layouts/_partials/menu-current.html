{{- /* The aria-current of a main menu entry: "page" on the entry of the page being rendered, "true" on one that holds it, "" elsewhere. IsMenuCurrent and HasMenuCurrent only know entries linked to a page, and the demos and the README write url = "posts/", which links none. So the resolved url is compared too, trailing slash aside, and holds every page under it. The home url holds every page, so it is only ever "page". A parent holds what any of its children holds. Both sides are compared decoded, as relLangURL leaves "catégorie/" and RelPermalink escapes it. A url with a query or a fragment is not compared. */ -}} {{- $page := .page -}} {{- $entry := .entry -}} {{- $state := "" -}} {{- $here := $page.RelPermalink -}} {{- with (try (urls.Parse $here)).Value }}{{ $here = .Path }}{{ end -}} {{- $here = strings.TrimSuffix "/" $here -}} {{- $url := "" -}} {{- $local := false -}} {{- if $entry.URL -}} {{- with (try (urls.Parse (partial "menu-url.html" $entry))).Value -}} {{- if not (or .Scheme .Host .RawQuery .Fragment) -}} {{- $url = strings.TrimSuffix "/" .Path -}} {{- $local = true -}} {{- end -}} {{- end -}} {{- end -}} {{- if or ($page.IsMenuCurrent "main" $entry) (and $local (eq $url $here)) -}} {{- $state = "page" -}} {{- else if $page.HasMenuCurrent "main" $entry -}} {{- $state = "true" -}} {{- else if and $local (ne $url (strings.TrimSuffix "/" (urls.Parse site.Home.RelPermalink).Path)) (hasPrefix $here (printf "%s/" $url)) -}} {{- $state = "true" -}} {{- else -}} {{- range $entry.Children -}} {{- if partial "menu-current.html" (dict "page" $page "entry" .) -}} {{- $state = "true" -}} {{- end -}} {{- end -}} {{- end -}} {{- return $state -}}
