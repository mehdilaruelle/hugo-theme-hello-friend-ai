{{- /* aria-current of a main menu entry: "page", "true" when it holds the page, or "". */ -}} {{- $page := .page -}} {{- $entry := .entry -}} {{- $state := "" -}} {{- $here := $page.RelPermalink -}} {{- with (try (urls.Parse $here)).Value }}{{ $here = .Path }}{{ end -}} {{- $here = strings.TrimSuffix "/" $here -}} {{- $url := "" -}} {{- $local := false -}} {{- if $entry.URL -}} {{- with (try (urls.Parse (partial "menu-url.html" $entry))).Value -}} {{- if not (or .Scheme .Host) -}} {{- $url = strings.TrimSuffix "/" .Path -}} {{- $local = true -}} {{- end -}} {{- end -}} {{- end -}} {{- if or ($page.IsMenuCurrent "main" $entry) (and $local (eq $url $here)) -}} {{- $state = "page" -}} {{- else if $page.HasMenuCurrent "main" $entry -}} {{- $state = "true" -}} {{- else if and $local (ne $url (strings.TrimSuffix "/" (urls.Parse site.Home.RelPermalink).Path)) (hasPrefix $here (printf "%s/" $url)) -}} {{- $state = "true" -}} {{- else -}} {{- range $entry.Children -}} {{- if partial "menu-current.html" (dict "page" $page "entry" .) -}} {{- $state = "true" -}} {{- end -}} {{- end -}} {{- end -}} {{- return $state -}}
