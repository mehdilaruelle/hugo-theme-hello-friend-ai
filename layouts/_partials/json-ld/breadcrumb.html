{{- /* BreadcrumbList for a single page in a section; the last item has no url. */ -}} {{- $section := "" -}} {{- if .IsPage -}} {{- with .Parent -}}{{- if not .IsHome -}}{{- $section = . -}}{{- end -}}{{- end -}} {{- end -}} {{- with $section -}} {{- $items := slice -}} {{- $i := 0 -}} {{- /* An ancestor that is never rendered has no URL to give, and Google wants one on every crumb but the last: it is left out, and the positions close up. */ -}} {{- range where $.Ancestors.Reverse "Permalink" "!=" "" -}} {{- $i = add $i 1 -}} {{- $name := "" -}} {{- if .IsHome -}} {{- $name = partial "i18n.html" (dict "key" "home" "fallback" "Home") -}} {{- else -}} {{- $name = .LinkTitle -}} {{- end -}} {{- $items = $items | append (dict "@type" "ListItem" "position" $i "name" $name "item" .Permalink) -}} {{- end -}} {{- $items = $items | append (dict "@type" "ListItem" "position" (add $i 1) "name" (partial "title-plain.html" $)) -}} {{- $trail := dict "@context" "https://schema.org" "@type" "BreadcrumbList" "itemListElement" $items -}} {{- end -}}
