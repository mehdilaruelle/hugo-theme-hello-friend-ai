(in (slice "index" "_index") .File.BaseFileName) (not (hasSuffix $.base "/")) -}} {{- $bundle = printf "%s/" (replaceRE `\.[^./]*$` "" $.base) -}} {{- end -}} {{- if ne $bundle $base -}} {{- range .Resources }}{{ $names = $names | append .Name }}{{ end -}} {{- end -}} {{- end -}} {{- /* A replacement template reads $ as a group. */ -}} {{- $b := replace $base "$" "$$" -}} {{- $bb := replace $bundle "$" "$$" -}} {{- $md := `[^/#?<)\s:][^)\s:]*` -}} {{- $def := `[^/#?<>\s:][^>\s:]*` -}} {{- $dq := `[^/#?"\s:][^"\s:]*` -}} {{- $sq := `[^/#?'\s:][^'\s:]*` -}} {{- $dp := `(?: {0,3}> ?)* {0,3}` -}} {{- /* A definition's label, not a footnote's, and what may follow its destination: a title. */ -}} {{- $label := `\[[^\]^][^\]]*\]:` -}} {{- $title := `(?:[ \t]+(?:"[^"]*"|'[^']*'|\([^)]*\)))?[ \t]*$` -}} {{- $lines := slice -}} {{- $code := partial "md/code-lines.html" .text -}} {{- range $n, $raw := split .text "\n" -}} {{- $line := $raw -}} {{- if not (index $code $n) -}} {{- $out := "" -}} {{- range partial "md/spans.html" $line -}} {{- $p := .text -}} {{- if not .code -}} {{- range $name := $names -}} {{- $q := printf `(?:\./)?%s` (replaceRE `([\\.+*?()|\[\]{}^$])` `\$1` $name) -}} {{- $to := print $bb (replace $name "$" "$$") -}} {{- $p = replaceRE (printf `\]\(%s([?#)\s])` $q) (printf `](%s${1}` $to) $p -}} {{- $p = replaceRE (printf `\]\(<%s([?#>])` $q) (printf `](<%s${1}` $to) $p -}} {{- $p = replaceRE (printf `^(%s%s[ \t]*]*)?>?%s)` $dp $label $q $title) (printf `${1}%s${2}` $to) $p -}} {{- $p = replaceRE (printf `\]\(<([^/#?<>:\n][^<>:\n]*)>` (printf `](<%s${1}>` $b) $p -}} {{- $p = replaceRE (printf `^(%s%s[ \t]*?%s)` $dp $label $def $title) (printf `${1}%s${2}${3}` $b) $p -}} {{- $p = replaceRE (printf $p -}} {{- end -}} {{- $line = $out -}} {{- end -}} {{- $lines = $lines | append $line -}} {{- end -}} {{- return delimit $lines "\n" -}}
