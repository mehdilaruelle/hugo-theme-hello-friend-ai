{{- /* The body as Markdown. A shortcode rendering its inner content sends it through Goldmark, whose typographer returns ’ and its kin: right in the HTML the shortcode produced, noise in a file read as text. Only those are decoded — < > & " are markup once decoded, and a code sample inside a shortcode is entitled to keep them written as entities. Code keeps them all, and a leading + becomes \+, not a list marker. */ -}} {{- $entities := dict "…" "…" "«" "«" "“" "“" "‘" "‘" "—" "—" "–" "–" "»" "»" "”" "”" "’" "’" "'" "'" "+" "+" -}} {{- $text := .RenderShortcodes -}} {{- $lines := slice -}} {{- $code := partial "md/code-lines.html" $text -}} {{- range $n, $raw := split $text "\n" -}} {{- $line := $raw -}} {{- if not (index $code $n) -}} {{- $line = replaceRE `^(\s*)+(\s|$)` `${1}\+${2}` $line -}} {{- $out := "" -}} {{- range partial "md/spans.html" $line -}} {{- $p := .text -}} {{- if not .code -}} {{- range $entity, $char := $entities }}{{ $p = replace $p $entity $char }}{{ end -}} {{- end -}} {{- $out = print $out $p -}} {{- end -}} {{- $line = $out -}} {{- end -}} {{- $lines = $lines | append $line -}} {{- end -}} {{- return delimit $lines "\n" -}}
