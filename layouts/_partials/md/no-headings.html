{{- /* Markdown with each heading outside code printed as a bold line. */ -}} {{- $lines := slice -}} {{- $code := partial "md/code-lines.html" . -}} {{- /* The line before, when it can be a setext heading's text: top-level prose. */ -}} {{- $text := "" -}} {{- range $n, $raw := split . "\n" -}} {{- $line := $raw -}} {{- if not (index $code $n) -}} {{- if and $text (findRE `^ {0,3}(?:=+|-+)[ \t]*\r?$` $raw 1) -}} {{- /* Its underline goes, and the line above it goes bold. */ -}} {{- $lines = first (sub (len $lines) 1) $lines | append (replaceRE `^ *(.*?)[ \t]*(\r?)$` `**${1}**${2}` $text) -}} {{- $text = "" }}{{ continue -}} {{- end -}} {{- /* One in a quote or a list item keeps its prefix. An empty one goes, or **** would draw a rule. CRLF keeps its \r after the closing **. */ -}} {{- $in := `^((?: {0,3}>[ \t]?)*(?: {0,3}(?:[-*+]|\d{1,9}[.)])[ \t]+)?) {0,3}#{1,6}` -}} {{- $line = replaceRE (print $in `(?:[ \t]+#+)?[ \t]*(\r?)$`) `${1}${2}` $line -}} {{- $line = replaceRE (print $in `(?:[ \t]+(.*?))??(?:[ \t]+#+)?[ \t]*(\r?)$`) `${1}**${2}**${3}` $line -}} {{- end -}} {{- $text = cond (and (not (index $code $n)) (strings.TrimSpace $raw) (not (findRE `^ {0,3}(?:[>#<]|[-*+](?:[ \t]|\r?$)|\d{1,9}[.)](?:[ \t]|\r?$)|(?:[-*_=][ \t]*)+\r?$)` $raw 1))) $raw "" -}} {{- $lines = $lines | append $line -}} {{- end -}} {{- return delimit $lines "\n" -}}
