{{- /* Markdown with each ATX heading outside a fence printed as a bold line. */ -}} {{- $lines := slice -}} {{- $fence := "" -}} {{- range split . "\n" -}} {{- $line := . -}} {{- if $fence -}} {{- with findRESubmatch "^[\\s>]*(`{3,}|~{3,})\\s*$" $line 1 -}} {{- $m := index . 0 1 -}} {{- if and (eq (substr $m 0 1) (substr $fence 0 1)) (ge (len $m) (len $fence)) }}{{ $fence = "" }}{{ end -}} {{- end -}} {{- else -}} {{- with findRESubmatch "^[\\s>]*(?:(?:[-*+]|\\d{1,9}[.)])\\s+)?(`{3,}|~{3,})(.*)$" $line 1 -}} {{- $m := index . 0 -}} {{- if not (and (hasPrefix (index $m 1) "`") (strings.Contains (index $m 2) "`")) }}{{ $fence = index $m 1 }}{{ end -}} {{- end -}} {{- if not $fence -}} {{- $line = replaceRE `^ {0,3}#{1,6}(?:[ \t]+(.*?))??(?:[ \t]+#+)?[ \t]*$` `**${1}**` $line -}} {{- end -}} {{- end -}} {{- $lines = $lines | append $line -}} {{- end -}} {{- return delimit $lines "\n" -}}
