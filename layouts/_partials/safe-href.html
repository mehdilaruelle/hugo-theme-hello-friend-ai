{{- /* A site-written URL for an href (social, menu, logo); "" when refused. html/template turns any scheme but http(s)/mailto into #ZgotmplZ, so xmpp:, tel:, sgnl:... are marked safe. Only schemes that run code are refused; blanks are dropped first, as a browser does ("java script:"). */ -}} {{- $url := printf "%v" . -}} {{- $out := $url -}} {{- with findRESubmatch `^([a-zA-Z][a-zA-Z0-9+.-]*):` (replaceRE `[\x00-\x20]` "" $url) 1 -}} {{- if in (slice "javascript" "vbscript" "data") (lower (index . 0 1)) -}} {{- warnidf "url-scheme-refused" "%q: a link that runs code is not written." $url -}} {{- $out = "" -}} {{- else -}} {{- $out = safeURL $url -}} {{- end -}} {{- end -}} {{- return $out -}}
