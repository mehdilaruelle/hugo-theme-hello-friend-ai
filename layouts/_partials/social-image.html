{{- /* The picture a social card shows, and how to describe it. Returns url, alt, owned and the picture's size; url is empty when there is nothing to show. One place, because opengraph.html and twitter_cards.html both ask, and a card whose halves disagree shows one picture and names another. Order: the page's images, its cover, the site's images, then ogImage. An SVG is skipped at every step, so the chain falls through to one that will render. alt and owned are set only for a picture belonging to this page: json-ld.html takes it only then, since one file asserted as the subject of two hundred articles is worse data than none. They are set from $img rather than beside it, because images = [""] -- the README's own sample -- passes `with` and then falls through. */ -}} {{- $img := "" -}} {{- $alt := "" -}} {{- $owned := false -}} {{- with .Params.images -}} {{- if not (partial "is-svg.html" (index . 0)) -}}{{- $img = index . 0 -}}{{- end -}} {{- end -}} {{- with $img -}} {{- $alt = partial "title.html" $ -}} {{- $owned = true -}} {{- end -}} {{- if not $img -}} {{- with .Params.cover -}} {{- if not (partial "is-svg.html" .) -}} {{- $img = . -}} {{- $alt = or $.Params.coverCaption (partial "title.html" $) -}} {{- $owned = true -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $img -}} {{- with site.Params.images -}} {{- if not (partial "is-svg.html" (index . 0)) -}}{{- $img = index . 0 -}}{{- end -}} {{- end -}} {{- end -}} {{- if not $img -}} {{- with site.Params.ogImage -}} {{- if not (partial "is-svg.html" .) -}}{{- $img = . -}}{{- end -}} {{- end -}} {{- end -}} {{- /* Resolved as a resource so the URL names a file Hugo wrote; a remote one is left alone. The size comes off that same resource, for og:image:width and og:image:height: without them Facebook and LinkedIn hold the first render back until they have fetched and measured the file. A format that will not give its dimensions up answers 0, which is worse asserted than omitted. */ -}} {{- $url := "" -}} {{- $width := 0 -}} {{- $height := 0 -}} {{- with $img -}} {{- $url = . -}} {{- if not (partial "is-remote.html" $url) -}} {{- with or ($.Resources.GetMatch $url) (resources.Get $url) -}} {{- $url = .Permalink -}} {{- if reflect.IsImageResourceWithMeta . -}} {{- if gt .Width 0 -}} {{- $width = .Width -}} {{- $height = .Height -}} {{- end -}} {{- end -}} {{- else -}} {{- $url = urls.JoinPath site.BaseURL $url -}} {{- end -}} {{- end -}} {{- end -}} {{- with $alt -}} {{- /* coverCaption is Markdown and an attribute is text. htmlUnescape because plainify strips tags but leaves the entities markdownify made, so an apostrophe reached the reader as a literal "’". TrimSpace because chomp takes newlines, and the flattening turns one into a space. */ -}} {{- $alt = . | markdownify | plainify | htmlUnescape | replaceRE "\\s+" " " | strings.TrimSpace -}} {{- end -}} {{- return dict "url" $url "alt" $alt "owned" $owned "width" $width "height" $height -}}
