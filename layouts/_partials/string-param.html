{{- /* (dict "value" v "name" "cover"): v when it is a string, else "" with a warning rather than a template error further down. "want" defaults to "a path". */ -}} {{- $out := "" -}} {{- $type := printf "%T" .value -}} {{- if eq $type "string" -}} {{- $out = .value -}} {{- else if ne $type "" -}} {{- warnidf (printf "%s-shape" (replace .name "." "-")) "%s: expected %s, got %T. It is ignored." .name (.want | default "a path") .value -}} {{- end -}} {{- return $out -}}
