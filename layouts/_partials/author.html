{{- /* The page's author name, else the site's; "" when there is none. A string, a map with a name key, or a list of either, joined with commas. Not cond: .name fails on a string. */ -}} {{- $author := (partial "site-author.html" .).name -}} {{- with .Params.author -}} {{- if reflect.IsMap . -}} {{- $author = .name -}} {{- else if reflect.IsSlice . -}} {{- $names := slice -}} {{- range . -}} {{- $name := . -}} {{- if reflect.IsMap . }}{{ $name = .name }}{{ end -}} {{- with $name }}{{ $names = $names | append (printf "%v" .) }}{{ end -}} {{- end -}} {{- $author = delimit $names ", " -}} {{- else -}} {{- $author = . -}} {{- end -}} {{- end -}} {{- return $author -}}
