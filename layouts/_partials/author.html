{{- /* Page author, else site author, else "": a string, a {name} map or a list. Not cond: .name fails on a string. */ -}} {{- $author := (partialCached "site-author.html" .).name -}} {{- with .Params.author -}} {{- if reflect.IsMap . -}} {{- $author = .name -}} {{- else if reflect.IsSlice . -}} {{- $names := slice -}} {{- range . -}} {{- $name := . -}} {{- if reflect.IsMap . }}{{ $name = .name }}{{ end -}} {{- with $name }}{{ $names = $names | append (printf "%v" .) }}{{ end -}} {{- end -}} {{- $author = delimit $names ", " -}} {{- else -}} {{- $author = . -}} {{- end -}} {{- end -}} {{- return $author -}}
