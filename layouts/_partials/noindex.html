{{- /* Whether this page is kept out of a search engine's index. Returns a boolean. noindex in the front matter is the site's decision; a search page is the theme's. searchable: false only keeps a page out of the theme's own JSON index, so four languages' worth of empty results list went into the sitemap. By layout rather than by path, because a page becomes a search page by asking for one. */ -}} {{- $noindex := false -}} {{- /* with covers the falsey spellings; the slice covers the ones a template reads as true because they are non-empty strings. Without it noindex: "false" asked to be indexed and was hidden, here and in the sitemap. */ -}} {{- with .Params.noindex -}} {{- if not (in (slice "false" "0") (lower (printf "%v" .))) -}}{{- $noindex = true -}}{{- end -}} {{- end -}} {{- if eq .Layout "search" -}}{{- $noindex = true -}}{{- end -}} {{- return $noindex -}}
