{{- /* The rules the site's params set, concatenated onto the stylesheet by head.html rather than written as style attributes, which a strict style-src refuses. */ -}} {{- $css := slice -}} {{- $cursor := "" -}} {{- if partial "switch.html" site.Params.Logo.logoCursorDisabled -}} {{- $cursor = "visibility:hidden;" -}} {{- end -}} {{- with site.Params.Logo.logoCursorColor -}} {{- $cursor = printf "%sbackground-color:%v;" $cursor . -}} {{- end -}} {{- with site.Params.Logo.logoCursorAnimate -}} {{- $cursor = printf "%sanimation-duration:%v;" $cursor . -}} {{- end -}} {{- with $cursor -}} {{- $css = $css | append (printf ".logo__cursor{%s}" .) -}} {{- end -}} {{- /* The dark colour wins wherever the palette is dark: the system's choice unless the toggle says light, or the toggle's. The rule above stays the light one, and the stylesheet's default when logoCursorColor is unset. */ -}} {{- with site.Params.Logo.logoCursorColorDark -}} {{- $dark := printf ".logo__cursor{background-color:%v}" . -}} {{- $css = $css | append (printf "@media (prefers-color-scheme:dark){:root:not([data-theme=light]) %s}:root[data-theme=dark] %s" $dark $dark) -}} {{- end -}} {{- with site.Params.backgroundImage -}} {{- $bg := printf "--background-image:url(%q);" (urls.JoinPath site.BaseURL .) -}} {{- with site.Params.backgroundImageLight -}} {{- $bg = printf "%s--background-image-light:url(%q);" $bg (urls.JoinPath site.BaseURL .) -}} {{- end -}} {{- $css = $css | append (printf ".background-image{%s}" $bg) -}} {{- end -}} {{- with partial "portrait-width.html" . -}} {{- $css = $css | append (printf "img.circle.portrait{max-width:%s}" .) -}} {{- end -}} {{- return delimit $css "" -}}
