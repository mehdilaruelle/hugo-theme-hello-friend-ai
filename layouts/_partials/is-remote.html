{{- /* Reports whether a URL points somewhere other than this site, and is the one place that decides it. A scheme is the obvious case. A protocol-relative URL has none — urls.Parse gives "//cdn.example.com/x.css" an empty Scheme and a Host of cdn.example.com — so testing the scheme alone reads it as a local path and prefixes the site's base URL, which breaks it. {{ if not (partial "is-remote.html" $src) }} {{ $src = urls.JoinPath site.BaseURL $src }} {{ end }} */ -}} {{- /* One that does not parse, "https://x/100%.png", is read by its shape. */ -}} {{- $remote := false -}} {{- with try (urls.Parse .) -}} {{- if .Err -}} {{- $remote = gt (len (findRE `^(?:[a-zA-Z][a-zA-Z0-9+.-]*:|//)` $ 1)) 0 -}} {{- else -}} {{- $remote = or (ne .Value.Scheme "") (ne .Value.Host "") -}} {{- end -}} {{- end -}} {{- return $remote -}}
