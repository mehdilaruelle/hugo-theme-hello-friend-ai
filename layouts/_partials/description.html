{{- /* What a page says about itself, in one place. Returns the empty string when it has nothing of its own: the site-wide fallback belongs in description-meta.html, since repeating one sentence in the structured data of every page is the duplication this chain exists to avoid. A taxonomy or term page with no summary is described by naming its term, rather than by the site description that would make each look like a copy of the next when each holds a different set of posts. */ -}} {{- $description := "" -}} {{- if and .IsHome (not .Description) -}} {{- /* homeSubtitle is the line under the title. A front page with a description of its own says what the site is for at the length a search result gives it, so that wins. */ -}} {{- $description = .Site.Params.homeSubtitle | plainify | htmlUnescape -}} {{- else if .Description -}} {{- $description = .Description -}} {{- else if and (eq .Kind "taxonomy") (not .Summary) -}} {{- /* Through i18n.html rather than i18n: with enableMissingTranslationPlaceholders on, a missing key renders "[i18n] key" rather than the empty string, so the test below never fires and the placeholder is what gets published. */ -}} {{- $description = partial "i18n.html" (dict "key" (printf "taxonomyDescription%s" (title .Data.Singular)) "fallback" "") -}} {{- if not $description -}} {{- $description = partial "i18n.html" (dict "key" "taxonomyDescription" "fallback" "") -}} {{- end -}} {{- else if and (eq .Kind "term") (not .Summary) -}} {{- /* .Data.Singular is Hugo's own word for the taxonomy, so it is English whatever the page is written in. One string per taxonomy instead. */ -}} {{- $description = partial "i18n.html" (dict "key" (printf "termDescription%s" (title .Data.Singular)) "args" (dict "Term" .Data.Term) "fallback" "") -}} {{- if not $description -}} {{- $description = partial "i18n.html" (dict "key" "termDescription" "args" (dict "Term" .Data.Term) "fallback" "") -}} {{- end -}} {{- else -}} {{- /* ~155 characters shown. htmlUnescape again: truncate escapes plain text. */ -}} {{- $description = partial "flatten.html" .Summary | truncate 160 "" | htmlUnescape -}} {{- end -}} {{- return $description -}}
