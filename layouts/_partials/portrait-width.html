{{- /* site.Params.portrait.maxWidth as a CSS length: a bare number, quoted or not, is pixels, and 0 is unset. */ -}} {{- $width := "" -}} {{- with site.Params.portrait.maxWidth -}} {{- $width = printf "%v" . | strings.TrimSpace -}} {{- if findRE `^[0-9]+(\.[0-9]+)?$` $width -}} {{- $px := float $width -}} {{- $width = cond (gt $px 0) (printf "%vpx" $px) "" -}} {{- else if not (findRE `^[0-9]+(\.[0-9]+)?(px|rem|em|ch|vw|%)$` $width) -}} {{- warnidf "portrait-width-invalid" "params.portrait.maxWidth: %q is not a CSS length such as 120px or 8rem, so the default width is kept." $width -}} {{- $width = "" -}} {{- end -}} {{- end -}} {{- return $width -}}
