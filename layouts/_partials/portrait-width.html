{{- /* site.Params.portrait.maxWidth as a CSS length: a bare number, quoted or not, is pixels, and 0 is unset. Any unit, or a calc()/min()/max()/clamp()/var() holding nothing that ends a value, a rule or the stylesheet. */ -}} {{- $width := "" -}} {{- with site.Params.portrait.maxWidth -}} {{- $width = printf "%v" . | strings.TrimSpace -}} {{- $num := `(?:[0-9]+(?:\.[0-9]+)?|\.[0-9]+)` -}} {{- $unit := `(?:px|r?em|r?ex|r?ch|r?lh|r?cap|r?ic|[sld]?v(?:w|h|i|b|min|max)|cq(?:w|h|i|b|min|max)|pt|pc|cm|mm|q|in|%)` -}} {{- if findRE (printf `^%s$` $num) $width -}} {{- $px := float $width -}} {{- $width = cond (gt $px 0) (printf "%vpx" $px) "" -}} {{- else if not (or (findRE (printf `(?i)^%s%s$` $num $unit) $width) (and (findRE `(?i)^(?:calc|min|max|clamp|var)\(.+\)$` $width) (not (findRE `[;{}<>"'\\\r\n]|/\*` $width)))) -}} {{- warnidf "portrait-width-invalid" "params.portrait.maxWidth: %q is not a CSS length such as 120px, 8rem or clamp(80px, 20vw, 160px), so the default width is kept." $width -}} {{- $width = "" -}} {{- end -}} {{- end -}} {{- return $width -}}
