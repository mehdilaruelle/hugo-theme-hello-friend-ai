{{- /* The pages a list page shows, in display order, before pagination. */ -}} {{- $pages := .Pages -}} {{- with .Data.Pages -}}{{- $pages = . -}}{{- end -}} {{- /* Newest first: Hugo's default order puts weight before date. */ -}} {{- if eq $.Kind "taxonomy" -}}{{- $pages = $pages.ByTitle -}}{{- else -}}{{- $pages = $pages.ByDate.Reverse -}}{{- end -}} {{- $pages = partial "listable.html" $pages -}} {{- /* A term with nothing left to list and no page of its own is left out. */ -}} {{- if eq $.Kind "taxonomy" -}} {{- $kept := first 0 $pages -}} {{- range $pages }}{{ if or .File (partial "list-pages.html" .) }}{{ $kept = $kept | append . }}{{ end }}{{ end -}} {{- $pages = $kept -}} {{- end -}} {{- return $pages -}}
