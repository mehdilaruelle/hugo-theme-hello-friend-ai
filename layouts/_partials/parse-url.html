{{- /* urls.Parse, or "" if unparsable. A % that starts no escape ("img/50%off.png") reads as itself (%25) instead of ending the build. */ -}} {{- $s := replace (replaceRE `%([0-9A-Fa-f]{2})` "\x00$1" (printf "%v" .)) "%" "%25" -}} {{- $u := "" -}} {{- with try (urls.Parse (replace $s "\x00" "%")) }}{{ if not .Err }}{{ $u = .Value }}{{ end }}{{ end -}} {{- return $u -}}
