{{- /* The URL this page names as its own, and which pager of a list it is. Returns a map with url, pager, and the neighbouring pagers' URLs as prev and next -- empty strings off a paginated list, so a caller can drop the tag. Every pager shares one .Permalink; naming it on page 2 and beyond would tell a search engine they are duplicates of page 1, which is the opposite of what pagination is for. The paginator is built only on the kinds list.html paginates: building it elsewhere would write the /page/2/ output files for nothing. Here because head.html, opengraph.html and json-ld.html all need the answer. */ -}} {{- $canonical := .Permalink -}} {{- $pager := 1 -}} {{- $prev := "" -}} {{- $next := "" -}} {{- if in (slice "section" "taxonomy" "term") .Kind -}} {{- /* Rebased on .Permalink, the only one canonifyURLs leaves its subpath. */ -}} {{- $relDir := replaceRE `[^/]*$` "" .RelPermalink -}} {{- $absDir := replaceRE `[^/]*$` "" .Permalink -}} {{- with partial "paginator.html" . -}} {{- $abs := dict -}} {{- range $k, $p := dict "url" . "prev" .Prev "next" .Next -}} {{- with $p -}} {{- $abs = merge $abs (dict $k (cond (hasPrefix .URL $relDir) (printf "%s%s" $absDir (strings.TrimPrefix $relDir .URL)) (absURL .URL))) -}} {{- end -}} {{- end -}} {{- $canonical = $abs.url -}} {{- $pager = .PageNumber -}} {{- $prev = $abs.prev | default "" -}} {{- $next = $abs.next | default "" -}} {{- end -}} {{- end -}} {{- return dict "url" $canonical "pager" $pager "prev" $prev "next" $next -}}
