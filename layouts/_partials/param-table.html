{{- /* site.Params. (lowercased) if it is a table, else an empty one. A plain value (logo = "img/logo.png", as other themes take it) ended the build; it now warns once (partialCached) and reads as unset. */ -}} {{- $value := index site.Params . -}} {{- $table := dict -}} {{- if reflect.IsMap $value -}} {{- $table = $value -}} {{- else if $value -}} {{- warnidf (printf "%s-shape" .) "params.%s: expected a table such as [params.%s], got a %T. Ignored." . . $value -}} {{- end -}} {{- return $table -}}
