{{- /* One content image. Resizing needs the file to resolve as a resource: a page bundle, or assets/. A file under static/ is not one, so it goes out whole and unmeasured. Takes src, alt, position, style, page, loading, fetchpriority, maxWidth, sizes, fill. */ -}} {{- $src := .src -}} {{- $alt := .alt | default "" -}} {{- $page := .page -}} {{- $loading := .loading | default "lazy" -}} {{- /* No default: marking every picture high priority prioritises none. */ -}} {{- $fetchpriority := .fetchpriority -}} {{- /* The measure of an article; a caller showing a fixed size passes its own. */ -}} {{- $sizes := .sizes | default site.Params.imageSizes | default "(max-width: 800px) 100vw, 800px" -}} {{- $remote := partial "is-remote.html" $src -}} {{- $img := "" -}} {{- if not $remote -}} {{- with $page.Resources.GetMatch $src -}} {{- $img = . -}} {{- else -}} {{- with resources.Get $src -}}{{- $img = . -}}{{- end -}} {{- end -}} {{- end -}} {{- /* Measuring and resizing are separate questions: an SVG answers neither and asking aborts the build, HEIC carries dimensions but cannot be resized, and Hugo returns 0 where the file will not say -- width="0" collapses. */ -}} {{- $maxWidth := .maxWidth | default site.Params.imageMaxWidth | default 1400 -}} {{- $rendered := $img -}} {{- $measurable := false -}} {{- if $img -}}{{- if reflect.IsImageResourceWithMeta $img -}} {{- $measurable = gt $img.Width 0 -}} {{- end -}}{{- end -}} {{- /* Only widths below the source: upscaling invents detail and pays for it. */ -}} {{- $widths := slice 480 800 1200 -}} {{- $srcset := slice -}} {{- $wantsFill := ne (.fill | default "") "" -}} {{- /* A GIF keeps its animation unless a fixed shape is asked for. */ -}} {{- $processable := and $img (reflect.IsImageResourceProcessable $img) (or $wantsFill (ne $img.MediaType.SubType "gif")) -}} {{- /* Capping fixes one axis only, and a square box stretches the other. Fill crops from the centre, like the object-fit: cover it replaces. */ -}} {{- if and $processable $wantsFill -}} {{- /* Processable answers from the header; the pixels are only read here, and a file Go cannot decode would otherwise abort the whole build. */ -}} {{- $cut := try ($img.Fill (printf "%s q85 webp" .fill)) -}} {{- with $cut.Err -}} {{- warnidf "image-fill-failed" "image.html: %s could not be cropped to %s and is served whole (%s)" $src $.fill . -}} {{- else -}} {{- $rendered = $cut.Value -}} {{- end -}} {{- else if $processable -}} {{- $ceiling := cond (gt $img.Width $maxWidth) $maxWidth $img.Width -}} {{- $offered := slice -}} {{- range $widths -}} {{- if lt . $ceiling -}}{{- $offered = $offered | append . -}}{{- end -}} {{- end -}} {{- $offered = $offered | append $ceiling -}} {{- /* Source width, one option: nothing to shrink, so no re-encode. */ -}} {{- if and (eq (len $offered) 1) (eq $ceiling $img.Width) -}} {{- $srcset = slice -}} {{- else -}} {{- range $offered -}} {{- $copy := $img.Resize (printf "%dx q85 webp" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $copy.RelPermalink $copy.Width) -}} {{- if eq . $ceiling -}}{{- $rendered = $copy -}}{{- end -}} {{- end -}} {{- if eq (len $offered) 1 -}}{{- $srcset = slice -}}{{- end -}} {{- end -}} {{- end -}} {{- $url := $src -}} {{- with $rendered -}} {{- $url = .RelPermalink -}} {{- else -}} {{- if not $remote -}}{{- $url = urls.JoinPath site.BaseURL $src -}}{{- end -}} {{- end -}} {{- if $src -}} {{- end -}}
