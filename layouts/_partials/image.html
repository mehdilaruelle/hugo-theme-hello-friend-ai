{{- /* One content image, resized and measured when src resolves to a page or assets/ resource. */ -}} {{- $src := .src -}} {{- $alt := .alt | default "" -}} {{- $page := .page -}} {{- $loading := .loading | default "lazy" -}} {{- $fetchpriority := .fetchpriority -}} {{- $measure := "800px" -}} {{- with partial "content-width.html" . -}} {{- if not (strings.HasSuffix . "%") -}}{{- $measure = . -}}{{- end -}} {{- end -}} {{- $sizes := .sizes | default site.Params.imageSizes | default (printf "(max-width: %s) 100vw, %s" $measure $measure) -}} {{- $remote := or .asWritten (partial "is-remote.html" $src) -}} {{- $img := "" -}} {{- if not $remote -}} {{- $img = partial "find-resource.html" (dict "page" $page "src" $src) -}} {{- end -}} {{- /* Pixels, bare or in px as the theme's other widths are written. */ -}} {{- $maxWidth := 0 -}} {{- with .maxWidth | default site.Params.imageMaxWidth -}} {{- with findRESubmatch `^\s*([0-9]+(?:\.[0-9]+)?)\s*(?:px)?\s*$` (printf "%v" .) 1 -}} {{- $maxWidth = int (float (index . 0 1)) -}} {{- else -}} {{- warnidf "image-max-width-shape" "imageMaxWidth: expected a number of pixels such as 1200, got %q. 1400 is used." (printf "%v" .) -}} {{- end -}} {{- end -}} {{- if le $maxWidth 0 }}{{ $maxWidth = 1400 }}{{ end -}} {{- $rendered := $img -}} {{- $measurable := false -}} {{- $svgWidth := 0 -}} {{- $svgHeight := 0 -}} {{- if $img -}}{{- if reflect.IsImageResourceWithMeta $img -}} {{- $measurable = gt $img.Width 0 -}} {{- else if eq $img.MediaType.SubType "svg" -}} {{- $tag := "" -}} {{- with findRE `]*>` $img.Content 1 -}}{{- $tag = index . 0 -}}{{- end -}} {{- $num := `([-+]?(?:\d+(?:\.\d*)?|\.\d+)(?:[eE][-+]?\d+)?)` -}} {{- $px := `\s%s\s*=\s*["']\s*%s\s*(?:px)?\s*["']` -}} {{- $absolute := `\s%s\s*=\s*["'][^"'%%]*["']` -}} {{- $w := findRESubmatch (printf $px "width" $num) $tag 1 -}} {{- $h := findRESubmatch (printf $px "height" $num) $tag 1 -}} {{- if and $w $h -}} {{- $svgWidth = int (math.Round (float (index $w 0 1))) -}} {{- $svgHeight = int (math.Round (float (index $h 0 1))) -}} {{- else if not (or (findRE (printf $absolute "width") $tag) (findRE (printf $absolute "height") $tag)) -}} {{- $n := printf `\s*%s` $num -}} {{- with findRESubmatch (printf `\sviewBox\s*=\s*["']%s[\s,]%s[\s,]%s[\s,]%s\s*["']` $n $n $n $n) $tag 1 -}} {{- $vw := float (index . 0 3) -}} {{- $vh := float (index . 0 4) -}} {{- if and (gt $vw 0) (gt $vh 0) -}} {{- $svgWidth = 1400 -}} {{- $svgHeight = int (math.Round (div (mul 1400 $vh) $vw)) -}} {{- end -}} {{- end -}} {{- end -}} {{- if or (le $svgWidth 0) (le $svgHeight 0) -}}{{- $svgWidth = 0 -}}{{- end -}} {{- end -}}{{- end -}} {{- $widths := slice 480 800 1200 -}} {{- $srcset := slice -}} {{- $wantsFill := ne (.fill | default "") "" -}} {{- $processable := and $img (reflect.IsImageResourceProcessable $img) (or $wantsFill (ne $img.MediaType.SubType "gif")) -}} {{- /* Since Hugo 0.168, a picture whose extension names another format (a JPEG saved as .png) answers 0 for its width and height, yet still resizes: the ceiling below came out 0 and the original was served whole and unmeasured. A copy made from it knows its own size, so it is asked. */ -}} {{- $natural := 0 -}} {{- $probe := "" -}} {{- if $processable -}} {{- $natural = $img.Width -}} {{- if le $natural 0 -}} {{- with try ($img.Process "webp") -}} {{- if and (not .Err) (gt .Value.Width 0) -}} {{- $probe = .Value -}} {{- $natural = $probe.Width -}} {{- $measurable = true -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if and $processable $wantsFill -}} {{- $cut := try ($img.Fill (printf "%s q85 webp" .fill)) -}} {{- with $cut.Err -}} {{- warnidf "image-fill-failed" "image.html: %s could not be cropped to %s and is served whole (%s)" $src $.fill . -}} {{- else -}} {{- $rendered = $cut.Value -}} {{- end -}} {{- else if $processable -}} {{- $ceiling := cond (gt $natural $maxWidth) $maxWidth $natural -}} {{- $offered := slice -}} {{- range $widths -}} {{- if lt . $ceiling -}}{{- $offered = $offered | append . -}}{{- end -}} {{- end -}} {{- $offered = $offered | append $ceiling -}} {{- if and (eq (len $offered) 1) (eq $ceiling $natural) -}} {{- $srcset = slice -}} {{- else -}} {{- range $offered -}} {{- $copy := try ($img.Resize (printf "%dx q85 webp" .)) -}} {{- with $copy.Err -}} {{- warnidf "image-resize-failed" "image.html: %s could not be resized and is served whole (%s)" $src . -}} {{- $srcset = slice -}} {{- break -}} {{- end -}} {{- $srcset = $srcset | append (printf "%s %dw" $copy.Value.RelPermalink $copy.Value.Width) -}} {{- if eq . $ceiling -}}{{- $rendered = $copy.Value -}}{{- end -}} {{- end -}} {{- if le (len $srcset) 1 -}}{{- $srcset = slice -}}{{- end -}} {{- end -}} {{- end -}} {{- $url := $src -}} {{- with $rendered -}} {{- $url = .RelPermalink -}} {{- else with .fallback -}} {{- $url = . -}} {{- /* A % that escapes nothing, "100%.png", is one itself. */ -}} {{- if $.asWritten -}}{{- $url = replace . "%" "%25" | replaceRE "%25([0-9A-Fa-f]{2})" "%$1" -}}{{- end -}} {{- else -}} {{- if not $remote -}}{{- $url = partialCached "abs-url.html" $src (printf "%v" $src) -}}{{- end -}} {{- end -}} {{- $attrs := .attrs | default dict -}} {{- $class := delimit (slice (.position | default "") ($attrs.class | default "")) " " | strings.TrimSpace -}} {{- $own := slice "src" "srcset" "loading" "decoding" "fetchpriority" "alt" "class" "style" "width" "height" "sizes" -}} {{- $width := 0 -}} {{- $height := 0 -}} {{- if $measurable -}}{{- $width = $rendered.Width -}}{{- $height = $rendered.Height -}} {{- /* Served whole, a picture measured through its copy still answers 0. */ -}} {{- with and (not $width) $probe -}}{{- $width = .Width -}}{{- $height = .Height -}}{{- end -}} {{- else if $svgWidth -}}{{- $width = $svgWidth -}}{{- $height = $svgHeight -}}{{- end -}} {{- /* A caller's pixel width or height wins; alone, the other follows the ratio. */ -}} {{- $px := `^\s*([0-9]+(?:\.[0-9]+)?)\s*(?:px)?\s*$` -}} {{- $w := 0 -}} {{- $h := 0 -}} {{- with findRESubmatch $px (string ($attrs.width | default "")) 1 -}}{{- $w = int (math.Round (float (index . 0 1))) -}}{{- end -}} {{- with findRESubmatch $px (string ($attrs.height | default "")) 1 -}}{{- $h = int (math.Round (float (index . 0 1))) -}}{{- end -}} {{- if and $w (not $h) $width -}}{{- $h = int (math.Max 1 (math.Round (div (mul $w $height) (float $width)))) -}}{{- end -}} {{- if and $h (not $w) $height -}}{{- $w = int (math.Max 1 (math.Round (div (mul $h $width) (float $height)))) -}}{{- end -}} {{- if or $w $h -}}{{- $width = $w -}}{{- $height = $h -}}{{- end -}} {{- with $attrs.sizes -}}{{- $sizes = . -}}{{- end -}} {{- /* alt arrives as HTML: entity-escaped text, maybe with tags. plainify strips the tags; safeHTML keeps the entities from being escaped again. Hugo 0.169 made plainify return a string unless given template.HTML, and "<b>" reached the reader as "&lt;b&gt;". The attribute escaper still strips tags and escapes quotes from an HTML value. */ -}} {{- if $src -}} {{- end -}}
