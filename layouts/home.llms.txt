{{- $desc := .Description | default (partial "flatten.html" (site.Params.homeSubtitle | default site.Params.description | default "")) -}}
{{- /* Not a list item: everything in a list here is a page. */ -}}
{{- $s := partial "md/sections.html" . -}}
{{- $total := add (len $s.posts) (len $s.other) -}}
{{- $limit := int (float (site.Params.llmsFullLimit | default 0)) -}}
{{- $full := "" -}}
{{- with .OutputFormats.Get "llmsfull" -}}
  {{- $full = printf "%s %s" (cond (and (gt $limit 0) (lt $limit $total)) (partial "i18n.html" (dict "key" "llmsFullAtFirst" "args" (dict "Kept" $limit) "fallback" (printf "The full text of the first %d pages below:" $limit))) (partial "i18n.html" (dict "key" "llmsFullAt" "fallback" "The full text of every page below:"))) .Permalink -}}
{{- end -}}
{{ partial "md/header.html" (dict "page" . "title" site.Title "desc" $desc "note" true "llms" true "after" $full) }}
