{{- /* Hugo's own highlighting, wrapped so the copy button and the language label have something to be positioned against that does not scroll: an absolutely positioned box whose containing block *is* the scroller travels with the content, so they would slide onto the code. Same pattern as .table-wrap in render-table.html, and pre keeps its own overflow and tabindex. transform.Highlight is what Hugo calls with no hook, so the markup inside the wrapper is unchanged. The trailing newline is put back because Goldmark strips it from .Inner before a hook sees it and the default path does not -- without it the last line of every block, and of the clipboard text, lost its newline. */ -}} {{- $opts := .Options -}} {{- if not (or (isset $opts "lineanchors") (isset $opts "lineAnchors")) -}} {{- $opts = merge $opts (dict "lineAnchors" (printf "hl-%d" .Ordinal)) -}} {{- end -}} {{- $class := "" -}} {{- $attrs := "" -}} {{- range $k, $v := .Attributes -}} {{- with printf "%v" $v -}} {{- if eq $k "class" -}} {{- $class = printf " %s" (htmlEscape .) -}} {{- else -}} {{- $attrs = printf "%s %s=\"%s\"" $attrs $k (htmlEscape .) -}} {{- end -}} {{- end -}} {{- end -}} {{- $html := transform.Highlight (printf "%s\n" .Inner) .Type $opts -}} {{- if or $class $attrs -}} {{- $html = replace $html `
` (printf `
` $class $attrs) 1 | safeHTML -}} {{- end -}}
{{ $html }}
