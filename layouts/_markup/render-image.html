{{- /* Markdown pictures through image.html; the first one of a page with no cover is fetched first. */ -}} {{- $dest := htmlUnescape .Destination -}} {{- $unescape := "\\\\([!-/:-@\\[-`{-~])" -}} {{- $src := $dest -}} {{- $asWritten := false -}} {{- with try (urls.Parse $dest) -}} {{- if .Err -}} {{- $asWritten = true -}} {{- else if and (not (partial "is-remote.html" $dest)) (not .Value.RawQuery) (not .Value.Fragment) -}} {{- $src = .Value.Path -}} {{- end -}} {{- end -}} {{- $cover := and (eq (printf "%T" .Page.Params.cover) "string") .Page.Params.cover -}} {{- $first := and (eq .Ordinal 0) .Page.IsPage (not $cover) (not (.Page.Store.Get "renderString")) -}} {{- $title := .Title | replaceRE $unescape "$1" | htmlUnescape -}} {{- $attrs := merge (.Attributes | default dict) (cond (ne $title "") (dict "title" $title) dict) -}} {{- partial "image.html" (dict "src" $src "fallback" $dest "asWritten" $asWritten "alt" (.PlainText | replaceRE $unescape "$1" | htmlEscape) "page" .PageInner "loading" (cond $first "eager" "lazy") "fetchpriority" (cond $first "high" "") "attrs" $attrs ) -}}
