{{- /* A picture written in Markdown goes through image.html like the shortcode's. The first picture of a page with no cover is likely the Largest Contentful Paint, so it is fetched at once; the rest are lazy. Resources are looked up in .PageInner, the bundle of an included page. */ -}} {{- /* Goldmark leaves entities in the destination and the title, numeric ones in the plain text, and backslash escapes in both. Only entities closed by ";" are decoded: htmlUnescape alone reads "?a=1®ion=eu" as "®". */ -}} {{- $entity := "&(#[0-9]{1,7};|#[xX][0-9a-fA-F]{1,6};|[a-zA-Z][a-zA-Z0-9]{1,31};)" -}} {{- $unescape := "\\\\([!-/:-@\\[-`{-~])" -}} {{- $dest := .Destination | replaceRE "&" "&" | replaceRE $entity "&$1" | htmlUnescape -}} {{- $src := $dest -}} {{- $asWritten := false -}} {{- /* Looked up by its decoded path; a url with a query or a fragment, or one that does not parse, "100%.png", stays as written. */ -}} {{- with try (urls.Parse $dest) -}} {{- if .Err -}} {{- $asWritten = true -}} {{- else if and (not (partial "is-remote.html" $dest)) (not .Value.RawQuery) (not .Value.Fragment) -}} {{- $src = .Value.Path -}} {{- end -}} {{- end -}} {{- $cover := and (eq (printf "%T" .Page.Params.cover) "string") .Page.Params.cover -}} {{- /* Ordinal restarts in every RenderString: faq.html flags its own. */ -}} {{- $first := and (eq .Ordinal 0) .Page.IsPage (not $cover) (not (.Page.Store.Get "renderString")) -}} {{- $title := .Title | replaceRE $unescape "$1" | replaceRE "&" "&" | replaceRE $entity "&$1" | htmlUnescape -}} {{- $attrs := merge (.Attributes | default dict) (cond (ne $title "") (dict "title" $title) dict) -}} {{- partial "image.html" (dict "src" $src "fallback" $dest "asWritten" $asWritten "alt" (.PlainText | replaceRE $unescape "$1" | htmlEscape | replaceRE "&(#[0-9]{1,7};|#[xX][0-9a-fA-F]{1,6};)" "&$1") "page" .PageInner "loading" (cond $first "eager" "lazy") "fetchpriority" (cond $first "high" "") "attrs" $attrs ) -}}
