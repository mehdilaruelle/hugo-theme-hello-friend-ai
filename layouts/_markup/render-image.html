{{- /* Markdown pictures through image.html; the first one of a page with no cover is fetched first. */ -}} {{- /* Only entities closed by ";" are decoded: htmlUnescape alone reads "?a=1®ion=eu" as "®". */ -}} {{- $entity := "&(#[0-9]{1,7};|#[xX][0-9a-fA-F]{1,6};|[a-zA-Z][a-zA-Z0-9]{1,31};)" -}} {{- $unescape := "\\\\([!-/:-@\\[-`{-~])" -}} {{- $dest := .Destination | replaceRE "&" "&" | replaceRE $entity "&$1" | htmlUnescape -}} {{- $src := $dest -}} {{- $asWritten := false -}} {{- with try (urls.Parse $dest) -}} {{- if .Err -}} {{- $asWritten = true -}} {{- else if and (not (partialCached "is-remote.html" $dest $dest)) (not .Value.RawQuery) (not .Value.Fragment) -}} {{- $src = .Value.Path -}} {{- end -}} {{- end -}} {{- $cover := and (eq (printf "%T" .Page.Params.cover) "string") .Page.Params.cover -}} {{- $first := and (eq .Ordinal 0) .Page.IsPage (not $cover) (not (.Page.Store.Get "renderString")) -}} {{- $title := .Title | replaceRE $unescape "$1" | replaceRE "&" "&" | replaceRE $entity "&$1" | htmlUnescape -}} {{- $attrs := merge (.Attributes | default dict) (cond (ne $title "") (dict "title" $title) dict) -}} {{- partial "image.html" (dict "src" $src "fallback" $dest "asWritten" $asWritten "alt" (.Text | plainify) "page" .PageInner "loading" (cond $first "eager" "lazy") "fetchpriority" (cond $first "high" "") "attrs" $attrs ) -}}
