{{- /* A grid of thumbnails, each a link to its full picture. The pictures are the page bundle's images, or those `match` names: a glob over the bundle, then over assets/ when the bundle has none that match. A file under static/ is not a resource and cannot be cropped, so it is not found here. Ordered by file name; alt and caption come from the resource's metadata in the front matter. */ -}} {{- $match := .Get "match" | default "" -}} {{- $columns := .Get "columns" | default 3 | int -}} {{- if or (lt $columns 2) (gt $columns 4) -}} {{- errorf "gallery: columns is 2, 3 or 4, not %d — %s" $columns .Position -}} {{- end -}} {{- $images := slice -}} {{- if $match -}} {{- $images = .Page.Resources.Match $match -}} {{- if not $images }}{{ $images = resources.Match $match }}{{ end -}} {{- else -}} {{- $images = .Page.Resources.ByType "image" -}} {{- end -}} {{- /* An SVG is an image to Hugo too, and cannot be cropped: it would sit in the grid at its own shape. Kept out, along with anything not a picture that a broad glob caught. */ -}} {{- $images = where (where $images "ResourceType" "image") "MediaType.SubType" "!=" "svg" -}} {{- $images = sort $images "Name" -}} {{- if not $images -}} {{- warnidf "gallery-empty" "gallery: no image found%s on %s — %s" (cond (ne $match "") (printf " for %q" $match) " in the page bundle") .Page.RelPermalink .Position -}} {{- else -}}
{{- range $images }} {{- /* A thumbnail is a link and needs a name. The file name is a poor one, and the last resort: a resource's own title is the file's path unless the front matter sets it. */ -}} {{- $alt := .Params.alt | default "" -}} {{- if and (not $alt) (ne .Title .Name) }}{{ $alt = .Title }}{{ end -}} {{- if not $alt }}{{ $alt = path.BaseName .Name | humanize }}{{ end }} {{- end }}
{{- end -}}
