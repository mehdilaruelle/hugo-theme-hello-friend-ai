{{- /* Thumbnails of the bundle's images (or `match`, then assets/), each linking to the full file. */ -}} {{- $match := .Get "match" | default "" -}} {{- $columns := .Get "columns" | default 3 | int -}} {{- if or (lt $columns 2) (gt $columns 4) -}} {{- errorf "gallery: columns is 2, 3 or 4, not %d — %s" $columns .Position -}} {{- end -}} {{- $images := slice -}} {{- if $match -}} {{- $images = .Page.Resources.Match $match -}} {{- if not $images }}{{ $images = resources.Match $match }}{{ end -}} {{- else -}} {{- $images = .Page.Resources.ByType "image" -}} {{- end -}} {{- /* SVGs cannot be cropped. */ -}} {{- $images = where (where $images "ResourceType" "image") "MediaType.SubType" "!=" "svg" -}} {{- /* A front-matter `name` replaces .Name, so the file is read from its URL. */ -}} {{- $files := slice -}} {{- range $images }}{{ $files = $files | append (dict "path" (urls.Parse .RelPermalink).Path "image" .) }}{{ end -}} {{- $images = sort $files "path" -}} {{- if not $images -}} {{- warnidf "gallery-empty" "gallery: no image found%s on %s — %s" (cond (ne $match "") (printf " for %q" $match) " in the page bundle") .Page.RelPermalink .Position -}} {{- else -}}
{{- range $images }}{{ $path := .path }}{{ with .image }} {{- /* The link needs a name: alt, then a set title (an unset one reads as the path once a name is set), then the file name. */ -}} {{- $alt := .Params.alt | default "" -}} {{- if and (not $alt) (ne .Title .Name) (not (strings.HasSuffix $path (print "/" .Title))) }}{{ $alt = .Title }}{{ end -}} {{- if not $alt }}{{ $alt = path.BaseName $path | humanize }}{{ end }} {{- end }}{{ end }}
{{- end -}}
