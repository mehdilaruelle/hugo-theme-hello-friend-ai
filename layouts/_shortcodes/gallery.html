{{- /* Thumbnails of the bundle's images (or `match`, then assets/), each linking to the full file. */ -}} {{- $match := .Get "match" | default "" -}} {{- $columns := .Get "columns" | default 3 | int -}} {{- if or (lt $columns 2) (gt $columns 4) -}} {{- errorf "gallery: columns is 2, 3 or 4, not %d — %s" $columns .Position -}} {{- end -}} {{- $images := slice -}} {{- if $match -}} {{- $images = .Page.Resources.Match $match -}} {{- if not $images }}{{ $images = resources.Match $match }}{{ end -}} {{- else -}} {{- $images = .Page.Resources.ByType "image" -}} {{- end -}} {{- /* SVGs cannot be cropped. */ -}} {{- $images = where (where $images "ResourceType" "image") "MediaType.SubType" "!=" "svg" -}} {{- $images = sort $images "Name" -}} {{- if not $images -}} {{- warnidf "gallery-empty" "gallery: no image found%s on %s — %s" (cond (ne $match "") (printf " for %q" $match) " in the page bundle") .Page.RelPermalink .Position -}} {{- else -}}
{{- range $images }} {{- /* The link needs a name: alt, then a set title, then the file name. */ -}} {{- $alt := .Params.alt | default "" -}} {{- if and (not $alt) (ne .Title .Name) }}{{ $alt = .Title }}{{ end -}} {{- if not $alt }}{{ $alt = path.BaseName .Name | humanize }}{{ end }} {{- end }}
{{- end -}}
